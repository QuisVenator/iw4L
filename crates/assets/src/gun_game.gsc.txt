gunGameConnect()
{
    level endon("game_ended");
    for (;;)
    {
        level waittill("connected", player);
        player.gunGameLevel = 0;
        player thread gunGameSpawn();
        player thread gunGameDeath();
        player thread gunGameAdvance();
    }
}

gunGameSpawn()
{
    self endon("disconnect");
    self endon("game_ended");
    for (;;)
    {
        self waittill("spawned_player");
        wait 0.1;
        self gunGameEquip();
    }
}

gunGameDeath()
{
    self endon("disconnect");
    self endon("game_ended");
    for (;;)
    {
        self waittill("death", attacker, means);
        if (!isdefined(attacker) || attacker == self || (isdefined(means) && (means == "MOD_SUICIDE" || means == "MOD_FALLING")))
        {
            if (self.gunGameLevel > 0)
            {
                self.gunGameLevel--;
                self iprintln("Gun Game: suicide demoted you to " + (self.gunGameLevel + 1) + " / " + level.gunGameWeapons.size);
            }
            else
            {
                self iprintln("Gun Game: suicide cannot demote below the first weapon");
            }
        }
        else if (isdefined(attacker) && isplayer(attacker) && attacker != self && isdefined(attacker.gunGameLevel))
        {
            if (isdefined(means) && means == "MOD_MELEE")
            {
                previousLevel = self.gunGameLevel;
                if (self.gunGameLevel > 0)
                {
                    self.gunGameLevel--;
                    self iprintln("Gun Game: melee kill demoted you to " + (self.gunGameLevel + 1) + " / " + level.gunGameWeapons.size);
                }
                attacker iprintln("Gun Game: you stay; opponent " + (previousLevel + 1) + " -> " + (self.gunGameLevel + 1));
            }
            else if (attacker.gunGameLevel < level.gunGameWeapons.size - 1)
            {
                attacker.gunGameLevel++;
                attacker gunGameEquip();
            }
            else
            {
                level thread maps\mp\gametypes\_gamelogic::endGame(attacker, &"MP_SCORE_LIMIT_REACHED");
            }
        }
    }
}

gunGameAdvance()
{
    self endon("disconnect");
    self endon("game_ended");
    for (;;)
    {
        self waittill("gungame_advance");
        if (!isdefined(self.gunGameLevel))
        {
            self.gunGameLevel = 0;
        }
        if (self.gunGameLevel < level.gunGameWeapons.size - 1)
        {
            self.gunGameLevel++;
            self gunGameEquip();
        }
        else
        {
            self iprintln("Gun Game: already at the final weapon");
        }
    }
}

gunGameEquip()
{
    weapon = level.gunGameWeapons[self.gunGameLevel];
    self takeallweapons();
    self giveweapon(weapon);
    self givemaxammo(weapon);
    self switchtoweapon(weapon);
    self iprintln("Gun Game: " + (self.gunGameLevel + 1) + " / " + level.gunGameWeapons.size + " " + weapon);
}
