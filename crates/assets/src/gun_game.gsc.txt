main()
{
    level.scoreLimitOverride = true;
    level.gunGameProgression = tolower(getdvar("scr_gg_progression"));
    if (level.gunGameProgression != "random")
    {
        level.gunGameProgression = "ordered";
    }
    level.gunGameRandomScope = tolower(getdvar("scr_gg_random_scope"));
    if (level.gunGameRandomScope != "perplayer")
    {
        level.gunGameRandomScope = "shared";
    }
    level.gunGameFinalWeapon = tolower(getdvar("scr_gg_final"));
    if (level.gunGameFinalWeapon == "")
    {
        level.gunGameFinalWeapon = "t6:weapon/hatchet_mp";
    }
    level.gunGamePool = [];
    level.gunGameWeapons = [];
    if (level.gunGameProgression == "random")
    {
        level.gunGamePool = gunGameParseWeapons(getdvar("scr_gg_pool"));
        if (level.gunGamePool.size == 0)
        {
            level.gunGamePool = gunGameFallbackWeapons();
        }
        if (level.gunGameRandomScope == "shared")
        {
            level.gunGameWeapons = gunGameRandomSequence();
        }
    }
    else
    {
        level.gunGameWeapons = gunGameParseWeapons(getdvar("scr_gg_weapons"));
        if (level.gunGameWeapons.size == 0)
        {
            level.gunGameWeapons = gunGameFallbackWeapons();
        }
    }
    level thread gunGameConnect();
}

gunGameFallbackWeapons()
{
    weapons = [];
    weapons[0] = "iw4:weapon/usp_mp";
    weapons[1] = "t6:weapon/hatchet_mp";
    return weapons;
}

gunGameParseWeapons(text)
{
    weapons = [];
    tokens = strtok(text, ",; ");
    for (i = 0; i < tokens.size; i++)
    {
        weapon = tolower(tokens[i]);
        if (weapon != "")
        {
            weapons[weapons.size] = weapon;
        }
    }
    return weapons;
}

gunGameRandomSequence()
{
    pool = [];
    for (i = 0; i < level.gunGamePool.size; i++)
    {
        if (level.gunGamePool[i] != level.gunGameFinalWeapon)
        {
            pool[pool.size] = level.gunGamePool[i];
        }
    }
    weapons = [];
    count = pool.size;
    while (count > 0)
    {
        index = randomint(count);
        weapons[weapons.size] = pool[index];
        count--;
        pool[index] = pool[count];
    }
    weapons[weapons.size] = level.gunGameFinalWeapon;
    return weapons;
}

gunGameLevelCount(player)
{
    if (isdefined(player.gunGameWeapons))
    {
        return player.gunGameWeapons.size;
    }
    return level.gunGameWeapons.size;
}

gunGameWeaponAt(player, index)
{
    if (isdefined(player.gunGameWeapons))
    {
        return player.gunGameWeapons[index];
    }
    return level.gunGameWeapons[index];
}

gunGameConnect()
{
    level endon("game_ended");
    for (;;)
    {
        level waittill("connected", player);
        player.gunGameLevel = 0;
        if (level.gunGameProgression == "random" && level.gunGameRandomScope == "perplayer")
        {
            player.gunGameWeapons = gunGameRandomSequence();
        }
        player thread gunGameSpawn();
        player thread gunGameDeath();
        player thread gunGameAdvance();
    }
}

gunGameSpawn()
{
    self endon("disconnect");
    self endon("game_ended");
    for (;;)
    {
        self waittill("spawned_player");
        wait 0.1;
        self gunGameEquip();
    }
}

gunGameDeath()
{
    self endon("disconnect");
    self endon("game_ended");
    for (;;)
    {
        self waittill("death", attacker, means);
        if (!isdefined(attacker) || attacker == self || (isdefined(means) && (means == "MOD_SUICIDE" || means == "MOD_FALLING")))
        {
            levelCount = gunGameLevelCount(self);
            if (self.gunGameLevel > 0)
            {
                self.gunGameLevel--;
                self iprintln("Gun Game: suicide demoted you to " + (self.gunGameLevel + 1) + " / " + levelCount);
            }
            else
            {
                self iprintln("Gun Game: suicide cannot demote below the first weapon");
            }
        }
        else if (isdefined(attacker) && isplayer(attacker) && attacker != self && isdefined(attacker.gunGameLevel))
        {
            if (isdefined(means) && means == "MOD_MELEE")
            {
                previousLevel = self.gunGameLevel;
                levelCount = gunGameLevelCount(self);
                if (self.gunGameLevel > 0)
                {
                    self.gunGameLevel--;
                    self iprintln("Gun Game: melee kill demoted you to " + (self.gunGameLevel + 1) + " / " + levelCount);
                }
                attacker iprintln("Gun Game: you stay; opponent " + (previousLevel + 1) + " -> " + (self.gunGameLevel + 1));
            }
            else if (attacker.gunGameLevel < gunGameLevelCount(attacker) - 1)
            {
                attacker.gunGameLevel++;
                attacker gunGameEquip();
            }
            else
            {
                level thread maps\mp\gametypes\_gamelogic::endGame(attacker, &"MP_SCORE_LIMIT_REACHED");
            }
        }
    }
}

gunGameAdvance()
{
    self endon("disconnect");
    self endon("game_ended");
    for (;;)
    {
        self waittill("gungame_advance");
        if (!isdefined(self.gunGameLevel))
        {
            self.gunGameLevel = 0;
        }
        if (self.gunGameLevel < gunGameLevelCount(self) - 1)
        {
            self.gunGameLevel++;
            self gunGameEquip();
        }
        else
        {
            self iprintln("Gun Game: already at the final weapon");
        }
    }
}

gunGameEquip()
{
    levelCount = gunGameLevelCount(self);
    if (levelCount <= 0)
    {
        return;
    }
    index = self.gunGameLevel;
    if (index >= levelCount)
    {
        index = levelCount - 1;
    }
    weapon = gunGameWeaponAt(self, index);
    self takeallweapons();
    self giveweapon(weapon);
    self givemaxammo(weapon);
    self switchtoweapon(weapon);
    self iprintln("Gun Game: " + (self.gunGameLevel + 1) + " / " + levelCount + " " + weapon);
}
